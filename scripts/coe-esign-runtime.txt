 const COE_SIGNATURE_BUCKET = 'coe-signatures'
 const COE_SIGNATURE_FINAL_STATES = ['signed','released']
 const COE_HASH_FIELDS = ['employeeName','employeeCode','employmentStatus','employmentStartDate','employmentEndDate','positionTitle','assignedDepartment','addressee','certificatePurpose','compensationStatement','additionalCertification','signatoryTitle']

 const coeEsc = value => String(value ?? '').replace(/[&<>"']/g, ch => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[ch]))
 const coeNormalizeRpcJson = value => Array.isArray(value) ? (value[0] ?? null) : value
 const coeUpdateLocalStatus = (recordId, status) => {
  try {
   if (typeof setCompanyDocumentRecords === 'function') {
    setCompanyDocumentRecords(prev => (prev || []).map(row => String(row.id) === String(recordId) ? { ...row, status } : row))
   }
  } catch(_) {}
 }
 const coeFormatSignedAt = value => {
  if (!value) return ''
  const date = new Date(value)
  if (Number.isNaN(date.getTime())) return String(value)
  return date.toLocaleString('en-PH', { timeZone:'Asia/Manila', month:'long', day:'numeric', year:'numeric', hour:'numeric', minute:'2-digit' })
 }
 const coeBlobToDataUrl = blob => new Promise((resolve, reject) => {
  const reader = new FileReader()
  reader.onload = () => resolve(String(reader.result || ''))
  reader.onerror = () => reject(reader.error || new Error('Unable to read signature image.'))
  reader.readAsDataURL(blob)
 })
 const coeDownloadSignatureBlob = async path => {
  const cleanPath = String(path || '').trim()
  if (!cleanPath) throw new Error('No e-signature image is attached to this COE.')
  const { data, error } = await supabase.storage.from(COE_SIGNATURE_BUCKET).download(cleanPath)
  if (error) throw error
  return data
 }
 const coeDownloadSignatureDataUrl = async path => coeBlobToDataUrl(await coeDownloadSignatureBlob(path))

 const coeGetHashPayload = ({ values = {}, record = null } = {}) => {
  const cf = values?.customFields || values || {}
  const fields = {}
  COE_HASH_FIELDS.forEach(key => { fields[key] = String(cf?.[key] ?? values?.[key] ?? '').trim() })
  return {
   formKey:CERTIFICATE_OF_EMPLOYMENT_FORM_KEY,
   documentRecordId:record?.id ? String(record.id) : '',
   referenceNo:String(record?.reference_no || record?.document_no || values?.documentNo || '').trim(),
   employeeId:String(record?.employee_id || values?.employeeId || '').trim(),
   fields
  }
 }
 const coeComputeDocumentHash = async ({ values = {}, record = null } = {}) => {
  if (!globalThis.crypto?.subtle) throw new Error('Secure browser hashing is unavailable. Please use the app over HTTPS in a modern browser.')
  const payload = JSON.stringify(coeGetHashPayload({ values, record }))
  const digest = await globalThis.crypto.subtle.digest('SHA-256', new TextEncoder().encode(payload))
  return Array.from(new Uint8Array(digest)).map(byte => byte.toString(16).padStart(2, '0')).join('')
 }

 const coeInvalidateChangedSignature = async ({ record, values, state, currentHash }) => {
  if (!record?.id || !state || !COE_SIGNATURE_FINAL_STATES.includes(String(state.state || '').toLowerCase())) return state
  if (!state.document_hash || state.document_hash === currentHash) return state
  const { data, error } = await supabase.rpc('invalidate_coe_signature_if_changed', {
   p_document_record_id:String(record.id),
   p_current_document_hash:currentHash
  })
  if (error) throw error
  const next = coeNormalizeRpcJson(data) || { ...state, state:'draft' }
  coeUpdateLocalStatus(record.id, 'Draft')
  return next
 }

 const coePrepareRenderContext = async ({ values, record, requireSigned = false } = {}) => {
  if (!record?.id) {
   if (requireSigned) throw new Error('Save this COE first, then send it for e-signature before final printing or downloading.')
   return { signatureState:null, draftWatermark:true }
  }
  let state = await coeGetWorkflowState(record.id)
  const currentHash = await coeComputeDocumentHash({ values, record })
  state = await coeInvalidateChangedSignature({ record, values, state, currentHash })
  const finalState = COE_SIGNATURE_FINAL_STATES.includes(String(state?.state || '').toLowerCase())
  const hashMatches = !!state?.document_hash && state.document_hash === currentHash
  if (!finalState || !hashMatches) {
   if (requireSigned) {
    const label = String(state?.state || 'draft').replace(/_/g, ' ')
    throw new Error(`This COE is ${label} and cannot be issued as a final document until an authorized signatory signs the current version.`)
   }
   return { signatureState:state, draftWatermark:true }
  }
  const signatureImageDataUrl = await coeDownloadSignatureDataUrl(state.signed_signature_path)
  return { signatureState:{ ...state, signature_image_data_url:signatureImageDataUrl }, draftWatermark:false }
 }

   const canvas = modal.body.querySelector('[data-coe-signature-canvas]')
   const ctx = canvas.getContext('2d')
   ctx.lineWidth = 5
   ctx.lineCap = 'round'
   ctx.lineJoin = 'round'
   ctx.strokeStyle = '#111'
   const clearCanvas = () => ctx.clearRect(0, 0, canvas.width, canvas.height)
   const drawBlob = async blob => {
    const url = URL.createObjectURL(blob)
    try {
     const image = new Image()
     await new Promise((resolve, reject) => { image.onload = resolve; image.onerror = reject; image.src = url })
     clearCanvas()
     const scale = Math.min(canvas.width / image.width, canvas.height / image.height)
     const width = image.width * scale
     const height = image.height * scale
     ctx.drawImage(image, (canvas.width - width) / 2, (canvas.height - height) / 2, width, height)
    } finally { URL.revokeObjectURL(url) }
   }
   if (profile.signature_profile_path) {
    try { await drawBlob(await coeDownloadSignatureBlob(profile.signature_profile_path)) } catch(_) {}
   }
   let drawing = false
   let last = null
   const point = event => {
    const rect = canvas.getBoundingClientRect()
    return { x:(event.clientX - rect.left) * (canvas.width / rect.width), y:(event.clientY - rect.top) * (canvas.height / rect.height) }
   }
   canvas.addEventListener('pointerdown', event => { drawing = true; last = point(event); canvas.setPointerCapture?.(event.pointerId) })
   canvas.addEventListener('pointermove', event => { if (!drawing) return; const next = point(event); ctx.beginPath(); ctx.moveTo(last.x,last.y); ctx.lineTo(next.x,next.y); ctx.stroke(); last = next })
   const stop = () => { drawing = false; last = null }
   canvas.addEventListener('pointerup', stop); canvas.addEventListener('pointercancel', stop); canvas.addEventListener('pointerleave', stop)
   modal.body.querySelector('[data-coe-signature-clear]').onclick = clearCanvas
   modal.body.querySelector('[data-coe-signature-file]').onchange = async event => {
    const file = event.target.files?.[0]
    if (!file) return
    if (file.size > 2 * 1024 * 1024) { showToast('Signature image must be 2 MB or smaller.', 'red'); return }
    if (!['image/png','image/jpeg','image/webp'].includes(file.type)) { showToast('Use PNG, JPEG, or WEBP for the signature.', 'red'); return }
    try { await drawBlob(file) } catch(error) { showToast('Unable to load signature image: ' + (error?.message || error), 'red') }
   }
   modal.body.querySelector('[data-coe-signature-save]').onclick = async event => {
    const button = event.currentTarget
    button.disabled = true
    try {
     const blob = await new Promise(resolve => canvas.toBlob(resolve, 'image/png', 0.95))
     if (!blob || blob.size < 250) throw new Error('Draw or upload a signature before saving.')
     const path = `profiles/${profile.admin_user_id}/signature.png`
     const { error:uploadError } = await supabase.storage.from(COE_SIGNATURE_BUCKET).upload(path, blob, { contentType:'image/png', upsert:true, cacheControl:'0' })
     if (uploadError) throw uploadError
     const { error:profileError } = await supabase.rpc('set_my_coe_signature_profile', { p_signature_profile_path:path })
     if (profileError) throw profileError
     showToast('Your COE e-signature profile was saved.')
     modal.close()
    } catch(error) { showToast('Failed to save e-signature: ' + (error?.message || error), 'red') }
    finally { button.disabled = false }
   }
  } catch(error) { showToast('Unable to open e-signature profile: ' + (error?.message || error), 'red') }
 }

 const approveCoeSignature = async ({ record, values, state, modal }) => {
  let snapshotPath = ''
  try {
   const profile = await coeGetMySignatureProfile()
   if (!profile?.authorized) throw new Error('Your account is not an authorized COE signatory.')
   if (!profile?.signature_profile_path) {
    await openCoeSignatureProfile()
    throw new Error('Save your e-signature profile first, then reopen this COE to sign it.')
   }
   const documentHash = await coeComputeDocumentHash({ values, record })
   if (state?.document_hash && state.document_hash !== documentHash) {
    await coeInvalidateChangedSignature({ record, values, state, currentHash:documentHash })
    throw new Error('The COE changed after it was sent for signature. It has been invalidated and must be sent again.')
   }
   const signatureBlob = await coeDownloadSignatureBlob(profile.signature_profile_path)
   const randomPart = globalThis.crypto?.randomUUID ? globalThis.crypto.randomUUID() : `${Date.now()}-${Math.random().toString(36).slice(2)}`
   snapshotPath = `documents/${record.id}/${randomPart}.png`
   const { error:uploadError } = await supabase.storage.from(COE_SIGNATURE_BUCKET).upload(snapshotPath, signatureBlob, { contentType:'image/png', upsert:false, cacheControl:'0' })
   if (uploadError) throw uploadError
   const { data, error } = await supabase.rpc('approve_coe_signature_request', { p_document_record_id:String(record.id), p_signature_storage_path:snapshotPath, p_document_hash:documentHash })
   if (error) throw error
   coeUpdateLocalStatus(record.id, 'Signed')
   showToast('COE e-signed and approved successfully.')
   modal?.close?.()
   return coeNormalizeRpcJson(data)
  } catch(error) {
   if (snapshotPath) { try { await supabase.storage.from(COE_SIGNATURE_BUCKET).remove([snapshotPath]) } catch(_) {} }
   showToast('Unable to e-sign COE: ' + (error?.message || error), 'red')
   return null
  }
 }

 const openCoeSignatureWorkflow = async record => {
  try {
   if (!record?.id) throw new Error('Save the COE before starting the signature workflow.')
   const values = getSavedDocumentValues(record)
   let state = await coeGetWorkflowState(record.id)
   if (state && COE_SIGNATURE_FINAL_STATES.includes(String(state.state || '').toLowerCase())) {
    const currentHash = await coeComputeDocumentHash({ values, record })
    state = await coeInvalidateChangedSignature({ record, values, state, currentHash })
   }
   const status = String(state?.state || 'draft').toLowerCase()
   if (status === 'pending' && state?.can_sign) { await reviewCoeSignatureRequest(record.id); return }
   if (status === 'pending') {
    const modal = coeOpenModal({ title:'COE Pending Signature', subtitle:record.reference_no || record.document_no || '', width:'560px' })
    modal.body.innerHTML = `<div style="line-height:1.6">This COE is waiting for <strong>${coeEsc(state?.assigned_signatory_name || 'the authorized signatory')}</strong>.<br><span style="color:#777;font-size:12px">Sent ${coeEsc(coeFormatSignedAt(state?.requested_at))}</span></div>`
    return
   }
   if (status === 'signed' || status === 'released') {
    const modal = coeOpenModal({ title:'COE Signed', subtitle:record.reference_no || record.document_no || '', width:'580px' })
    modal.body.innerHTML = `<div style="line-height:1.65"><strong>Signed by:</strong> ${coeEsc(state?.signed_name || '')}<br><strong>Title:</strong> ${coeEsc(state?.signed_title || '')}<br><strong>Signed:</strong> ${coeEsc(coeFormatSignedAt(state?.signed_at))}<br><br>The clean final Print, PDF, and Word outputs are now unlocked.</div>${state?.can_request ? '<div style="margin-top:14px"><button data-coe-resend type="button" style="border:0;background:#666;color:#fff;border-radius:9px;padding:9px 12px;font-weight:900;cursor:pointer">SEND FOR NEW SIGNATURE</button></div>' : ''}`
    const resend = modal.body.querySelector('[data-coe-resend]')
    if (resend) resend.onclick = () => { modal.close(); sendCoeForSignature(record) }
    return
   }
   if (status === 'returned') {
    const modal = coeOpenModal({ title:'COE Returned for Correction', subtitle:record.reference_no || record.document_no || '', width:'620px' })
    modal.body.innerHTML = `<div style="background:#fff3f3;border:1px solid #efb5b5;border-radius:9px;padding:11px"><strong>Reason:</strong> ${coeEsc(state?.return_reason || 'Correction requested')}</div><div style="font-size:12px;color:#666;margin-top:10px">Correct the COE, save the updated record, then send it again for signature.</div><div style="margin-top:14px"><button data-coe-resend type="button" style="border:0;background:#CA1B1B;color:#fff;border-radius:9px;padding:10px 14px;font-weight:900;cursor:pointer">SEND UPDATED COE</button></div>`
    modal.body.querySelector('[data-coe-resend]').onclick = () => { modal.close(); sendCoeForSignature(record) }
    return
   }
   await sendCoeForSignature(record)
  } catch(error) { showToast('Unable to open COE signature workflow: ' + (error?.message || error), 'red') }
 }

 const coeLogFinalOutput = async (record, outputType) => {
  if (!record?.id) return
  try { await supabase.rpc('log_coe_signature_output', { p_document_record_id:String(record.id), p_output_type:String(outputType || 'OUTPUT').toUpperCase() }) } catch(_) {}
 }
